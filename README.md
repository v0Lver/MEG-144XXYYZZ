/**
 * @title NativeSoilDeed - Complete Land + Resource Ownership

 *   1. Physical land (GPS coordinates)
 *   2. Natural resources (minerals, water, timber, etc.)
 *   3. Tribal responsibilities (stewardship duties)
 *   4. Ancestral claims (lineage verification)
 */

import "@openzeppelin/contracts/token/ERC721/ERC721.sol";

contract NativeSoilDeed is ERC721 {
    
    struct LandParcel {
        uint256 tokenId;
        string tribeName;              // e.g., "Judah (Afro-American)"
        uint256 acres;
        string location;               // e.g., "Coffee County, Georgia"
        
        // GPS Boundaries (4 corners)
        int256 northEastLat;
        int256 northEastLon;
        int256 southWestLat;
        int256 southWestLon;
        
        // Natural Resources
        string[] resources;            // ["Cotton", "Timber", "Freshwater"]
        mapping(string => uint256) resourceQuantities; // Resource => Amount
        
        // Responsibilities
        string[] duties;               // ["Steward civil rights legacy", ...]
        
        // Ownership
        address owner;
        bytes32 lineageHash;           // SHA-256 of family tree
        uint256 registrationDate;
        
        // Legal
        bool isTransferable;
        bool isMortgageable;
        string[] legalDescriptions;    // Metes and bounds, etc.
    }
    
    mapping(uint256 => LandParcel) public parcels;
    mapping(address => uint256[]) public ownerParcels;
    mapping(string => uint256) public tribeToTotalAcres;
    
    uint256 public nextTokenId = 1;
    uint256 public totalAcresRegistered = 0;
    
    event DeedIssued(
        uint256 indexed tokenId,
        address indexed owner,
        string tribeName,
        uint256 acres,
        string location
    );
    
    event ResourcesAllocated(
        uint256 indexed tokenId,
        string[] resources
    );
    
    event DutiesAssigned(
        uint256 indexed tokenId,
        string[] duties
    );
    
    constructor() ERC721("Native Soil Deed", "NSD") {}
    
    /**
     * @dev Register land parcel with complete details
     */
    function registerDeed(
        string memory _tribeName,
        uint256 _acres,
        string memory _location,
        int256[4] memory _gpsBoundaries, // [NELat, NELon, SWLat, SWLon]
        string[] memory _resources,
        uint256[] memory _resourceQuantities,
        string[] memory _duties,
        bytes32 _lineageHash,
        bool _isTransferable,
        bool _isMortgageable,
        string[] memory _legalDescriptions
    ) external returns (uint256) {
        
        require(_resources.length == _resourceQuantities.length, "Resource mismatch");
        
        uint256 tokenId = nextTokenId++;
        
        LandParcel storage parcel = parcels[tokenId];
        parcel.tokenId = tokenId;
        parcel.tribeName = _tribeName;
        parcel.acres = _acres;
        parcel.location = _location;
        parcel.northEastLat = _gpsBoundaries[0];
        parcel.northEastLon = _gpsBoundaries[1];
        parcel.southWestLat = _gpsBoundaries[2];
        parcel.southWestLon = _gpsBoundaries[3];
        parcel.resources = _resources;
        
        // Map resource quantities
        for (uint i = 0; i < _resources.length; i++) {
            parcel.resourceQuantities[_resources[i]] = _resourceQuantities[i];
        }
        
        parcel.duties = _duties;
        parcel.owner = msg.sender;
        parcel.lineageHash = _lineageHash;
        parcel.registrationDate = block.timestamp;
        parcel.isTransferable = _isTransferable;
        parcel.isMortgageable = _isMortgageable;
        parcel.legalDescriptions = _legalDescriptions;
        
        // Mint NFT deed
        _mint(msg.sender, tokenId);
        ownerParcels[msg.sender].push(tokenId);
        tribeToTotalAcres[_tribeName] += _acres;
        totalAcresRegistered += _acres;
        
        emit DeedIssued(tokenId, msg.sender, _tribeName, _acres, _location);
        emit ResourcesAllocated(tokenId, _resources);
        emit DutiesAssigned(tokenId, _duties);
        
        return tokenId;
    }
    
    /**
     * @dev Extract resource from parcel (with limits)
     */
    function extractResource(
        uint256 _tokenId,
        string memory _resourceName,
        uint256 _amount
    ) external {
        require(ownerOf(_tokenId) == msg.sender, "Not owner");
        
        LandParcel storage parcel = parcels[_tokenId];
        require(
            parcel.resourceQuantities[_resourceName] >= _amount,
            "Insufficient resource"
        );
        
        parcel.resourceQuantities[_resourceName] -= _amount;
        
        // Could mint resource token here, transfer to owner
        // ResourceToken.mint(msg.sender, _amount);
    }
    
    /**
     * @dev Get parcel details
     */
    function getParcel(uint256 _tokenId) 
        external 
        view 
        returns (
            string memory tribeName,
            uint256 acres,
            string memory location,
            string[] memory resources,
            string[] memory duties,
            address owner
        ) 
    {
        LandParcel storage parcel = parcels[_tokenId];
        return (
            parcel.tribeName,
            parcel.acres,
            parcel.location,
            parcel.resources,
            parcel.duties,
            parcel.owner
        );
    }
    
    /**
     * @dev Get resource quantity
     */
    function getResourceQuantity(
        uint256 _tokenId,
        string memory _resourceName
    ) external view returns (uint256) {
        return parcels[_tokenId].resourceQuantities[_resourceName];
    }
    
    /**
     * @dev Get tribal statistics
     */
    function getTribeStats(string memory _tribeName) 
        external 
        view 
        returns (uint256 totalAcres) 
    {
        return tribeToTotalAcres[_tribeName];
    }
}
